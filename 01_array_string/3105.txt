class Solution {
    public int longestMonotonicSubarray(int[] nums) {
        int maxIncLen = 1;
        int maxDecLen = 1;
        int incLen = 1;
        int decLen = 1;

        for(int i = 1; i < nums.length; i++) {
            if (nums[i] > nums[i-1]) {
                incLen++;
                maxIncLen = Math.max(maxIncLen, incLen);
            } else {
                incLen = 1;
            }
            if (nums[i] < nums[i-1]) {
                decLen++;
                maxDecLen = Math.max(maxDecLen, decLen);
            } else {
                decLen = 1;
            }
        }

        return Math.max(maxIncLen, maxDecLen);
    }
}
