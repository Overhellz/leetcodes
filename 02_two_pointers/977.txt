class Solution {
    public int[] sortedSquares(int[] nums) {
        if (nums == null || nums.length == 0) {
            return nums;
        }

        int pos = nums.length - 1;
        int left = 0;
        int right  = nums.length - 1;
        int[] result = new int[nums.length];

        while(left <= right) {
            int leftSq = nums[left] * nums[left];
            int rightSq = nums[right] * nums[right];

            if (leftSq < rightSq) {
                result[pos] = rightSq;
                right--;
            } else {
                result[pos] = leftSq;
                left++
            }
            pos--;
        }

        return result;
    }
}
